** Turning builtin Led on

#+begin_src C
  # include <Arduino.h>
  void setup
  {
    pinMode(LEDBUILT, OUTPUT); // set the pin 13 to be in output
  }
  void loop ()
  {
    // Code that will excuated while the Arduino is running
    digitalWrite(LEDBUILTIN, HIGH);	// set a low voltage
    delay(1000);			// delay 1 sec
    digitalWrite(LEDBUILTIN, LOW);	// send a low voltage
    delay(1000);			// delay 1 sec
  }
#+end_src
